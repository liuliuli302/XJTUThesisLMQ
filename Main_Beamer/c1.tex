\section{研究背景与动机}

\begin{frame}{研究背景}
  \begin{itemize}
    \item 视频数据爆炸式增长，信息过载问题日益突出
    \item \textbf{视频摘要}：自动保留最关键的帧或片段，\\ 帮助用户快速把握视频内容
    \item 传统方法普遍依赖标注与任务特定训练，\\ 资源消耗大
  \end{itemize}
  % \vfill
  \begin{block}{研究趋势}
    多模态大语言模型（MLLMs）的快速发展，为\textbf{免训练（Training-free）}视频摘要提供了新范式：无需额外训练，仅依靠语言推理即可完成语义理解与关键片段筛选
  \end{block}
\end{frame}

\begin{frame}{现有方法的不足与本文动机}
  \begin{columns}[T]
    \begin{column}{0.6\textwidth}
      \textbf{现有方法局限}
      \begin{enumerate}
        \item 高层语义理解不足，难以把握视频主题
        \item 单一评分机制，难以覆盖多维评价目标
        \item 时序上下文利用有限，长视频一致性不足
      \end{enumerate}
    \end{column}
    \begin{column}{0.4\textwidth}
      \textbf{本文两条技术路线}
      \begin{enumerate}
        \item \textbf{层级语义建模}\\实现帧→场景→视频的逐级理解
        \item \textbf{多智能体协同推理}\\由 Planner + 多 Expert 联合完成多维评分
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{结构目录}
  \begin{enumerate}
    \item 研究背景与动机
    \item 相关理论基础
    \item 基于层级语义建模的免训练视频摘要方法
    \item 基于多智能体协同推理的视频摘要方法
    \item 结论与展望
    \item 质询问题回答
  \end{enumerate}
\end{frame}

\section{相关理论基础}

\begin{frame}{视频摘要任务概述}
  \begin{itemize}
    \item \textbf{任务定义}：给定视频 $\mathcal{V}=\{f_1,\ldots,f_T\}$，自动选择关键帧或关键片段生成摘要
    \item \textbf{常用数据集}：
      \begin{itemize}
        \item \textbf{SumMe}：包含 25 段用户生成视频，内容涵盖旅行、烹饪、极限运动和日常活动等开放域场景。每段视频由 15--18 位标注者分别给出代表性摘要片段。
        \item \textbf{TVSum}：包含 50 段 YouTube 视频，覆盖新闻、纪录片、教程、Vlog 等 10 个主题类别。每段视频提供 20 位标注者逐帧给出的重要性评分。
      \end{itemize}
    \item \textbf{评价指标}：$F_1$ 分数，以及 Spearman $\rho$、Kendall $\tau$ 秩相关系数
  \end{itemize}
\end{frame}

\begin{frame}{关键技术基础}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{大语言模型}
      \begin{itemize}
        \item Transformer 自注意力架构
        \item 提示工程 (Prompt Engineering)
        \item 多模态语义对齐
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{智能体与协作}
      \begin{itemize}
        \item 基于 LLM 的 Agent 决策
        \item 通过角色化提示实现职责分工
        \item 多智能体协同机制
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{阶段一：视频场景划分}
  \begin{itemize}
    \item 利用 \textbf{KTS 算法}对视频进行结构化分段
    \item 优化目标：
      \begin{equation*}
        \min_{\mathcal{B}} J(\mathcal{B})=\sum_{k=1}^{K}\mathrm{Var}(s_k)+\lambda K
      \end{equation*}
    \item 输出语义相近的场景单元 $\{s_1, s_2, \ldots, s_K\}$
  \end{itemize}
  \begin{figure}
    \centering
    \includegraphics[width=0.6\textwidth]{c3/model_kts.pdf}

    \end{column}
    \begin{column}[c]{0.48\textwidth}
      \begin{itemize}
        \item \textbf{帧级}：逐帧生成视觉描述
        \item \textbf{场景级}：聚合同一场景的文本信息
        \item \textbf{视频级}：生成全局主题摘要
      \end{itemize}
    \end{column}
  \end{columns}

\end{frame}

\begin{frame}{阶段三：全局主题约束的场景语义评分}

        \centering
        \includegraphics[width=\textwidth]{c3/model_llm_query.pdf}

    \begin{column}[c]{0.52\textwidth}
      \begin{itemize}
        \item 以视频级摘要 $Cap_v$ 作为全局参照
        \item 评估每个场景对整体主题的贡献程度
      \end{itemize}
      \begin{equation*}
        S_{\mathrm{LLM}}(k)=\mathrm{Norm}\big(\mathrm{LLM}(\Pi(Cap_v,Cap_s(k)))\big)
      \end{equation*}
    \end{column}
  \end{columns}
  \vfill
\end{frame}

        \item 场景内部：衡量帧与场景的一致性

        \item 二者融合得到最终帧级分数
      \end{itemize}
      \begin{equation*}
        F(p_j)=\alpha\,S_{\mathrm{LLM}}\big(\sigma(j)\big)\,S_{FS}(p_j)+S_{FV}(p_j)

\end{frame}

\begin{frame}{阶段一：视频结构化处理与段级描述生成}
  \vfill
  \begin{columns}[c,onlytextwidth]
    \begin{column}[c]{0.45\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\textwidth]{c4/model_c4_01.pdf}
      \end{figure}
    \end{column}
    \begin{column}[c]{0.52\textwidth}
      \begin{enumerate}
        \item \textbf{均匀分段}：将视频划分为 $N$ 个连续片段
        \item \textbf{片段内采样}：每段均匀选取代表帧
        \item \textbf{段级描述}：由 VLM 生成片段级文本描述
      \end{enumerate}
      \begin{equation*}
        Cap_s(k)=\mathrm{VLM}_{\mathrm{seg}}\big(I_k;\theta_{\mathrm{VLM}},\pi_{\mathrm{seg}}\big)
      \end{equation*}
    \end{column}
  \end{columns}
  \vfill
\end{frame}

\begin{frame}{阶段二：全局规划与多专家协同评分}
  \begin{figure}
    \centering
    \includegraphics[height=0.7\textheight]{c4/model_c4_02.pdf}
    \caption{全局规划与多专家协同评分流程}
  \end{figure}
\end{frame}

\begin{frame}{Planner Agent 与 Expert Agents}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Planner Agent 全局规划}
      \begin{itemize}
        \item 推断视频主题 $z$
        \item 生成全局摘要 $Cap_v$
        \item 分配专家权重 $\mathbf{w}$
        \item 融合各 Expert 证据给出主分数
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{四类 Expert Agent}
      \begin{itemize}
        \item \textbf{Story}：叙事推进
        \item \textbf{Visual}：视觉显著性
        \item \textbf{Emotion}：情绪表达
        \item \textbf{Information}：信息增量
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \textbf{分数融合：}
  \begin{equation*}
    S_{\mathrm{final}}(k) = S_{\mathrm{plan}}(k) + \sum_{m\in\{s,v,e,i\}} w_m \, S_m(k)
  \end{equation*}
\end{frame}

\begin{frame}{阶段三：记忆增强与帧级摘要生成}
  \begin{enumerate}
    \item \textbf{上下文记忆注入}——将前序片段描述拼接注入当前评分提示
      \begin{equation*}
        \mathcal{M}_k^{(h)}=\{Cap_s(\max(1,k{-}h)),\ldots,Cap_s(k{-}1)\}
      \end{equation*}
    \item \textbf{帧级分数映射}——段级分数回填至帧轴
      \begin{equation*}
        F(t)=S_{\mathrm{final}}\big(k(t)\big)
      \end{equation*}
    \item \textbf{背包优化}——0/1 动态规划选择最优片段集合
      \begin{equation*}
        \max_{\mathbf{x}} \sum_{j} x_j\,u_j \quad \text{s.t.}\quad \sum_{j} x_j\,l_j \le L,\; x_j\in\{0,1\}
      \end{equation*}
  \end{enumerate}
\end{frame}

\section{结论与展望}

\begin{frame}{主要贡献}
  \begin{enumerate}
    \item \textbf{层级语义建模的免训练视频摘要方法}
      \begin{itemize}
        \item 构建帧→场景→视频三级语义表示链路
        \item 引入全局主题约束与帧级语义增强，生成主题一致的摘要
      \end{itemize}
    \item \textbf{多智能体协同推理的视频摘要框架}
      \begin{itemize}
        \item 设计 Planner 全局规划 + 四类 Expert 多维协同评分机制
        \item 引入上下文记忆增强，提高摘要时序完整性
      \end{itemize}
    \item 两条技术路线在 SumMe / TVSum 上均取得有竞争力的结果，\\
          且无需额外训练与参数更新
  \end{enumerate}
\end{frame}

\begin{frame}{未来工作方向}
  \begin{enumerate}
    \item \textbf{提升长视频建模能力}：结合内容感知的自适应分段与动态检索式记忆机制，使推理单元更贴近真实事件边界，并增强长程语义一致性
    \item \textbf{面向用户需求的个性化摘要}：引入用户意图感知与动态专家权重调整机制，使摘要结果能够更好适配不同用户偏好与查询目标
  \end{enumerate}
\end{frame}

\section{质询回答}

\begin{frame}
  \begin{block}{质询问题1}
    层级语义建模方法中，KTS 算法正则化参数 $\lambda$ 的自适应调整具体策略是什么？不同视频类型（如运动类）的最优 $\lambda$ 范围有何差异？
  \end{block}
  \vfill
  \begin{itemize}
    \item \textbf{$\lambda$ 的确定方式}：KTS 算法的正则化参数 $\lambda$ 并未设计随视频内容自适应调整的机制，而是一个全局固定的经验参数，并在所有视频上统一使用
    \item \textbf{关于视频类型差异}：由于采用全局统一参数策略，未针对运动类、新闻类、Vlog 等不同视频类别分别标定最优 $\lambda$ 范围，因此目前没有不同类别对应的最优 $\lambda$ 差异分析
  \end{itemize}
  \vfill
\end{frame}

\begin{frame}
  \begin{block}{质询问题2}
    多智能体方法中，Planner Agent 动态分配专家权重的具体计算逻辑是什么？是否考虑了视频主题与专家维度的相关性量化？
  \end{block}
  \vfill
  \begin{itemize}
    \item \textbf{权重分配机制}：Planner Agent 基于全部片段描述，在单次 LLM 推理中同步输出视频主题标签、全局摘要 及四类 Expert 的归一化权重向量；不存在预设公式或外部参数，完全由 LLM 的上下文推理驱动
    \item \textbf{主题--维度相关性}：已通过提示工程设计内嵌考虑——系统提示要求 Planner 显式输出视频主题，并基于主题判断各维度的相对重要性；权重由 LLM 根据语义理解隐式推断，未引入显式的主题--维度相关性量化公式
  \end{itemize}
  \vfill
\end{frame}
