\section{研究背景与动机}

\begin{frame}{研究背景}
  \begin{itemize}
    \item 视频数据爆炸式增长，信息过载问题日益突出
    \item \textbf{视频摘要}：自动识别并保留最具代表性的帧或片段，\\ 支持用户快速理解视频主题
    \item 传统方法（监督/无监督/弱监督）依赖标注数据与任务特定训练，\\ 跨场景迁移能力有限
  \end{itemize}
  % \vfill
  \begin{block}{研究趋势}
    多模态大语言模型（MLLMs）的发展为\textbf{免训练 (Training-free)} 视频摘要提供了新范式——无需额外参数更新，利用语言推理能力完成语义分析与关键片段筛选
  \end{block}
\end{frame}

\begin{frame}{现有方法的不足与本文动机}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{现有方法局限}
      \begin{enumerate}
        \item 高层语义建模不足——难以刻画视频主题
        \item 单一评分器——无法兼顾多维评价目标
        \item 时序上下文利用有限——长视频一致性差
      \end{enumerate}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{本文两条技术路线}
      \begin{enumerate}
        \item \textbf{层级语义建模}\\帧→场景→视频的多级语义理解
        \item \textbf{多智能体协同推理}\\Planner + 多 Expert 多维度协同评分
      \end{enumerate}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{论文结构}
  \begin{enumerate}
    \item \textbf{第一章}：绪论——研究背景、研究现状与研究内容
    \item \textbf{第二章}：相关理论基础与技术原理
    \item \textbf{第三章}：基于层级语义建模的免训练视频摘要方法
    \item \textbf{第四章}：基于多智能体协同推理的视频摘要方法
    \item \textbf{第五章}：结论与展望
  \end{enumerate}

\end{frame}

\section{相关理论基础}

\begin{frame}{视频摘要任务概述}
  \begin{itemize}
    \item \textbf{任务定义}：给定视频 $\mathcal{V}=\{f_1,\ldots,f_T\}$，选择关键帧/片段集合构成摘要
    \item \textbf{常用数据集}：
      \begin{itemize}
        \item SumMe（25 段用户视频，评测 Max $F_1$）
        \item TVSum（50 段 YouTube 视频，评测 Avg $F_1$）
      \end{itemize}
    \item \textbf{评价指标}：$F_1$ 分数、Spearman $\rho$ / Kendall $\tau$ 秩相关系数
  \end{itemize}
\end{frame}

\begin{frame}{关键技术基础}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{大语言模型}
      \begin{itemize}
        \item Transformer 自注意力架构
        \item 提示工程 (Prompt Engineering)
        \item 多模态语义对齐
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{智能体与协作}
      \begin{itemize}
        \item 基于 LLM 的 Agent 决策
        \item 角色化提示实现职责分离
        \item 多智能体协同机制
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\section{层级语义建模的免训练视频摘要方法}

\begin{frame}{阶段一：视频场景划分}
  \begin{itemize}
    \item 采用 \textbf{KTS 算法}对视频进行结构化分段
    \item 优化目标：
      \begin{equation*}
        \min_{\mathcal{B}} J(\mathcal{B})=\sum_{k=1}^{K}\mathrm{Var}(s_k)+\lambda K
      \end{equation*}
    \item 将连续视频流组织为语义相近的场景单元 $\{s_1, s_2, \ldots, s_K\}$
  \end{itemize}
  \begin{figure}
    \centering
    \includegraphics[width=0.6\textwidth]{c3/model_kts.pdf}
  \end{figure}
\end{frame}

\begin{frame}{阶段二：多级语义描述生成}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \centering
      \includegraphics[width=0.95\textwidth]{c3/model_caption_aggregation.pdf}
      \captionof{figure}{多级视频语义描述生成流程}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{itemize}
        \item \textbf{帧级描述}：$Cap_f(t)=\mathrm{VLM}(I_t;\theta_{\mathrm{VLM}})$
        \item \textbf{场景级聚合}：$Cap_s(k)=\mathrm{LLM}_{\mathrm{scene}}(\mathcal{T}_k;\theta_{\mathrm{LLM}})$
        \item \textbf{视频级摘要}：$Cap_v=\mathrm{LLM}_{\mathrm{video}}(\{Cap_s(1),\ldots,Cap_s(K)\};\theta_{\mathrm{LLM}})$
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{阶段三：全局主题约束的场景语义评分}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\textwidth]{c3/model_llm_query.pdf}
      \end{figure}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item 联合输入场景描述 $Cap_s(k)$ 与视频级摘要 $Cap_v$
        \item LLM 评估该场景对整体主题的贡献
      \end{itemize}
      \begin{equation*}
        S_{\mathrm{LLM}}(k)=\mathrm{Norm}\big(\mathrm{LLM}(\Pi(Cap_v,Cap_s(k)))\big)
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{阶段四：帧级语义分数增强}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\textwidth]{c3/model_score_refinement.pdf}
      \end{figure}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item 帧--场景相关性：$S_{FS}(p_j)=\mathbf{z}_{f,j}^{\top}\mathbf{z}_{s,j}$
        \item 帧--视频相关性：$S_{FV}(p_j)=\mathbf{z}_{f,j}^{\top}\mathbf{z}_{v}$
        \item \textbf{最终帧级分数}：
      \end{itemize}
      \begin{equation*}
        F(p_j)=\alpha\,S_{\mathrm{LLM}}\big(\sigma(j)\big)\,S_{FS}(p_j)+S_{FV}(p_j)
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{阶段一：视频结构化处理与段级描述生成}
  \begin{columns}[T]
    \begin{column}{0.45\textwidth}
      \begin{figure}
        \centering
        \includegraphics[width=\textwidth]{c4/model_c4_01.pdf}
      \end{figure}
    \end{column}
    \begin{column}{0.52\textwidth}
      \begin{enumerate}
        \item \textbf{均匀分段}：将视频划分为 $N$ 个连续片段
        \item \textbf{片段内采样}：每段均匀选取代表帧
        \item \textbf{段级描述}：VLM 联合多帧输入生成段级文本
      \end{enumerate}
      \begin{equation*}
        Cap_s(k)=\mathrm{VLM}_{\mathrm{seg}}\big(I_k;\theta_{\mathrm{VLM}},\pi_{\mathrm{seg}}\big)
      \end{equation*}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{阶段二：全局规划与多专家协同评分}
  \begin{figure}
    \centering
    \includegraphics[height=0.7\textheight]{c4/model_c4_02.pdf}
    \caption{全局规划与多专家协同评分流程}
  \end{figure}
\end{frame}

\begin{frame}{Planner Agent 与 Expert Agents}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Planner Agent 全局规划}
      \begin{itemize}
        \item 推断视频主题 $z$
        \item 生成全局摘要 $Cap_v$
        \item 分配专家权重 $\mathbf{w}$
        \item 参考各 Expert 证据给出综合主分数
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{四类 Expert Agent}
      \begin{itemize}
        \item \textbf{Story}：叙事推进
        \item \textbf{Visual}：视觉显著性
        \item \textbf{Emotion}：情绪表达
        \item \textbf{Information}：信息增量
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \textbf{分数融合：}
  \begin{equation*}
    S_{\mathrm{final}}(k) = S_{\mathrm{plan}}(k) + \sum_{m\in\{s,v,e,i\}} w_m \, S_m(k)
  \end{equation*}
\end{frame}

\begin{frame}{阶段三：记忆增强与帧级摘要生成}
  \begin{enumerate}
    \item \textbf{上下文记忆注入}——将前序片段描述拼接注入当前评分提示
      \begin{equation*}
        \mathcal{M}_k^{(h)}=\{Cap_s(\max(1,k{-}h)),\ldots,Cap_s(k{-}1)\}
      \end{equation*}
    \item \textbf{帧级分数映射}——段级分数回填至帧轴
      \begin{equation*}
        F(t)=S_{\mathrm{final}}\big(k(t)\big)
      \end{equation*}
    \item \textbf{背包优化}——0/1 动态规划选择最优片段集合
      \begin{equation*}
        \max_{\mathbf{x}} \sum_{j} x_j\,u_j \quad \text{s.t.}\quad \sum_{j} x_j\,l_j \le L,\; x_j\in\{0,1\}
      \end{equation*}
  \end{enumerate}
\end{frame}

\section{结论与展望}

\begin{frame}{主要贡献}
  \begin{enumerate}
    \item \textbf{层级语义建模的免训练视频摘要方法}
      \begin{itemize}
        \item 帧→场景→视频三级语义描述 + 全局主题约束场景评分
        \item 帧级语义相关性增强，实现主题一致的摘要生成
      \end{itemize}
    \item \textbf{多智能体协同推理的视频摘要框架}
      \begin{itemize}
        \item Planner 全局规划 + 四类 Expert 多维度评分
        \item 上下文记忆增强 + 背包优化摘要构造
      \end{itemize}
    \item 两条路线在 SumMe / TVSum 数据集上取得有竞争力的性能，\\
          无需额外训练与参数更新
  \end{enumerate}
\end{frame}

      \textbf{第三章方法}
      \begin{itemize}
        \item SumMe: 48.1\% $F_1$
        \item TVSum: 61.1\% $F_1$
        \item 超越同类免训练方法\\ 6--10 个百分点
      \end{itemize}
